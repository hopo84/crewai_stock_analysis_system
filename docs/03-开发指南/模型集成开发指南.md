```python
# src/config/model_config.py
import os
from typing import Dict, Any, Optional
from enum import Enum

class ModelProvider(Enum):
    OPENAI = "openai"
    DEEPSEEK = "deepseek"

class ModelConfigManager:
    """模型配置管理器"""

    def __init__(self):
        self.provider = self._get_provider()
        self.config = self._load_config()

    def _get_provider(self) -> ModelProvider:
        """获取当前模型提供商"""
        provider_str = os.getenv('MODEL_PROVIDER', 'openai').lower()
        try:
            return ModelProvider(provider_str)
        except ValueError:
            raise ValueError(f"不支持的模型提供商: {provider_str}")

    def _load_config(self) -> Dict[str, Any]:
        """加载模型配置"""
        if self.provider == ModelProvider.OPENAI:
            return self._load_openai_config()
        elif self.provider == ModelProvider.DEEPSEEK:
            return self._load_deepseek_config()

    def get_chat_llm(self):
        """获取聊天模型实例"""
        from langchain_openai import ChatOpenAI

        return ChatOpenAI(
            model=self.config['chat_model'],
            api_key=self.config['api_key'],
            base_url=self.config['base_url'],
            max_tokens=self.config['max_tokens'],
            temperature=self.config['temperature']
        )

    def get_model_info(self) -> Dict[str, Any]:
        """获取当前模型信息"""
        return {
            'provider': self.provider.value,
            'chat_model': self.config.get('chat_model'),
            'manager_model': self.config.get('manager_model'),
            'max_tokens': self.config.get('max_tokens'),
            'temperature': self.config.get('temperature')
        }
```

### 3. 模型工厂模式

```python
# src/config/model_factory.py
from abc import ABC, abstractmethod
from langchain_openai import ChatOpenAI

class LLMFactory(ABC):
    """LLM工厂抽象基类"""

    @abstractmethod
    def create_chat_llm(self) -> ChatOpenAI:
        """创建聊天模型"""
        pass

    @abstractmethod
    def create_manager_llm(self) -> ChatOpenAI:
        """创建管理模型"""
        pass

class OpenAIFactory(LLMFactory):
    """OpenAI模型工厂"""

    def __init__(self, config: Dict[str, Any]):
        self.config = config

    def create_chat_llm(self) -> ChatOpenAI:
        return ChatOpenAI(
            model=self.config['chat_model'],
            openai_api_key=self.config['api_key'],
            openai_api_base=self.config['base_url'],
            max_tokens=self.config['max_tokens'],
            temperature=self.config['temperature']
        )

    def create_manager_llm(self) -> ChatOpenAI:
        return ChatOpenAI(
            model=self.config['manager_model'],
            openai_api_key=self.config['api_key'],
            openai_api_base=self.config['base_url'],
            max_tokens=self.config['max_tokens'],
            temperature=self.config['temperature']
        )

class DeepSeekFactory(LLMFactory):
    """DeepSeek模型工厂"""

    def __init__(self, config: Dict[str, Any]):
        self.config = config

    def create_chat_llm(self) -> ChatOpenAI:
        return ChatOpenAI(
            model=self.config['chat_model'],
            openai_api_key=self.config['api_key'],
            openai_api_base=self.config['base_url'],
            max_tokens=self.config['max_tokens'],
            temperature=self.config['temperature'],
            model_kwargs={'top_p': self.config.get('top_p', 0.95)}
        )

    def create_coder_llm(self) -> ChatOpenAI:
        """创建代码专用模型（DeepSeek特有）"""
        return ChatOpenAI(
            model=self.config['coder_model'],
            openai_api_key=self.config['api_key'],
            openai_api_base=self.config['base_url'],
            max_tokens=self.config['max_tokens'],
            temperature=self.config['temperature'],
            model_kwargs={'top_p': self.config.get('top_p', 0.95)}
```

## Agent集成开发

### 1. 模型感知的Agent基类

```python
# src/agents/base_agent.py
from crewai import Agent
from src.config.model_config import ModelConfigManager

class ModelAwareAgent:
    """模型感知的Agent基类"""

    def __init__(self):
        self.model_manager = ModelConfigManager()

    def create_agent_with_model(self, role: str, goal: str, backstory: str,
                               tools: list = None, model_type: str = "chat") -> Agent:
        """创建使用指定模型的Agent"""

        # 根据模型类型选择LLM
        if model_type == "chat":
            llm = self.model_manager.get_chat_llm()
        elif model_type == "manager":
            llm = self.model_manager.get_manager_llm()
        elif model_type == "coder" and hasattr(self.model_manager, 'get_coder_llm'):
            llm = self.model_manager.get_coder_llm()
        else:
            llm = self.model_manager.get_chat_llm()

        return Agent(
            role=role,
            goal=goal,
            backstory=backstory,
            tools=tools or [],
            llm=llm,
            verbose=True
        )
```

## 成本优化策略

### 1. 智能模型选择

```python
# src/utils/cost_optimizer.py
class CostOptimizer:
    """成本优化器"""

    def __init__(self, model_manager: ModelConfigManager):
        self.model_manager = model_manager

        # 模型成本映射（每1K tokens的成本，美元）
        self.cost_mapping = {
            'gpt-4o': 0.03,
            'gpt-4o-mini': 0.00015,
            'gpt-3.5-turbo': 0.001,
            'deepseek-chat': 0.00014,
            'deepseek-coder': 0.00014
        }

    def estimate_cost(self, model_name: str, input_tokens: int, output_tokens: int) -> float:
        """估算请求成本"""
        if model_name not in self.cost_mapping:
            return 0.0

        total_tokens = input_tokens + output_tokens
        return (total_tokens / 1000) * self.cost_mapping[model_name]

    def recommend_model_for_task(self, task_complexity: str, language: str = "zh") -> str:
        """为任务推荐最合适的模型"""

        # 中文任务优先推荐DeepSeek
        if language == "zh" and self.model_manager.provider == ModelProvider.DEEPSEEK:
            if task_complexity == "simple":
                return "deepseek-chat"
            elif task_complexity == "code":
                return "deepseek-coder"
            else:
                return "deepseek-chat"

        # 英文任务或OpenAI provider
        if task_complexity == "simple":
            return "gpt-3.5-turbo"
        elif task_complexity == "standard":
            return "gpt-4o-mini"
        else:
            return "gpt-4o"
```

### 2. 动态Token管理

```python
# src/utils/token_manager.py
import tiktoken

class TokenManager:
    """Token管理器"""

    def __init__(self, model_name: str):
        self.model_name = model_name
        self.encoder = tiktoken.encoding_for_model(model_name)

    def count_tokens(self, text: str) -> int:
        """计算文本token数量"""
        return len(self.encoder.encode(text))

    def truncate_text(self, text: str, max_tokens: int) -> str:
        """截断文本到指定token数量"""
        tokens = self.encoder.encode(text)
        if len(tokens) <= max_tokens:
            return text

        truncated_tokens = tokens[:max_tokens]
        return self.encoder.decode(truncated_tokens)

    def optimize_prompt(self, prompt: str, max_tokens: int) -> str:
        """优化prompt长度"""
        if self.count_tokens(prompt) <= max_tokens:
            return prompt

        # 简化策略：保留开头和结尾，压缩中间部分
        lines = prompt.split('\n')

        # 保留前3行和后3行
        if len(lines) > 6:
            start_lines = lines[:3]
            end_lines = lines[-3:]
            middle_summary = "[...内容已压缩...]"

            optimized_prompt = '\n'.join(start_lines + [middle_summary] + end_lines)

            if self.count_tokens(optimized_prompt) <= max_tokens:
                return optimized_prompt

        # 如果还是太长，直接截断
        return self.truncate_text(prompt, max_tokens)
```

```python
# src/utils/env_validator.py
class EnvironmentValidator:
    """环境配置验证器"""

    def __init__(self):
        self.required_vars = {
            'openai': [
                'OPENAI_API_KEY',
                'OPENAI_CHAT_MODEL'
            ],
            'deepseek': [
                'DEEPSEEK_API_KEY',
                'DEEPSEEK_CHAT_MODEL'
            ]
        }

    def validate_environment(self) -> Dict[str, Any]:
        """验证当前环境配置"""
        provider = os.getenv('MODEL_PROVIDER', 'openai').lower()

        if provider not in self.required_vars:
            return {
                'valid': False,
                'error': f'不支持的提供商: {provider}'
            }

        missing_vars = []
        for var in self.required_vars[provider]:
            if not os.getenv(var):
                missing_vars.append(var)

        if missing_vars:
            return {
                'valid': False,
                'error': f'缺少环境变量: {", ".join(missing_vars)}'
            }

        return {
            'valid': True,
            'provider': provider,
            'message': f'{provider} 配置验证通过'
        }
```

```python
# src/utils/model_monitor.py
import logging
from typing import Dict, Any
from datetime import datetime

class ModelMonitor:
    """模型使用监控"""

    def __init__(self):
        self.usage_stats = {
            'requests': 0,
            'total_tokens': 0,
            'total_cost': 0.0,
            'errors': 0,
            'start_time': datetime.now()
        }

        self.logger = logging.getLogger(__name__)

    def record_request(self, model_name: str, input_tokens: int,
                      output_tokens: int, cost: float, success: bool):
        """记录API请求"""
        self.usage_stats['requests'] += 1
        self.usage_stats['total_tokens'] += input_tokens + output_tokens
        self.usage_stats['total_cost'] += cost

        if not success:
            self.usage_stats['errors'] += 1

        self.logger.info(
            f"模型请求: {model_name}, "
            f"输入tokens: {input_tokens}, "
            f"输出tokens: {output_tokens}, "
            f"成本: ${cost:.6f}, "
            f"成功: {success}"
        )

    def get_usage_report(self) -> Dict[str, Any]:
        """获取使用报告"""
        runtime = datetime.now() - self.usage_stats['start_time']
        error_rate = (self.usage_stats['errors'] /
                     max(self.usage_stats['requests'], 1)) * 100

        return {
            'runtime_hours': runtime.total_seconds() / 3600,
            'total_requests': self.usage_stats['requests'],
            'total_tokens': self.usage_stats['total_tokens'],
            'total_cost': self.usage_stats['total_cost'],
            'error_rate': error_rate,
            'avg_cost_per_request': (self.usage_stats['total_cost'] /
                                   max(self.usage_stats['requests'], 1))
        }
```

### 1. 错误处理

```python
def robust_llm_call(llm, prompt: str, max_retries: int = 3) -> str:
    """健壮的LLM调用"""
    for attempt in range(max_retries):
        try:
            response = llm.invoke(prompt)
            return response.content

        except RateLimitError:
            wait_time = 2 ** attempt
            time.sleep(wait_time)
            continue

        except AuthenticationError:
            raise ValueError("API密钥无效或已过期")

        except Exception as e:
            if attempt == max_retries - 1:
                raise e
            time.sleep(1)

    raise Exception(f"LLM调用失败，已重试{max_retries}次")
```

```python
class HotReloadConfig:
    """支持热重载的配置管理"""

    def __init__(self):
        self._config = None
        self._last_reload = 0
        self.reload_interval = 60  # 60秒检查一次
```
